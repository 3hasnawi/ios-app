cmake_minimum_required(VERSION 3.20)
project(PayrollCompression C)
include(FetchContent)
FetchContent_Declare(apple_lzfse GIT_REPOSITORY https://github.com/lzfse/lzfse.git GIT_TAG e634ca58b4821d9f3d560cdc6df5dec02ffc93fd)
FetchContent_GetProperties(apple_lzfse)
if(NOT apple_lzfse_POPULATED)
  FetchContent_Populate(apple_lzfse)
endif()
file(GLOB codec_sources "${apple_lzfse_SOURCE_DIR}/src/*.c")
list(FILTER codec_sources EXCLUDE REGEX "lzfse_main.c$")
add_library(lzfse SHARED ${codec_sources})
target_compile_definitions(lzfse PRIVATE LZFSE_DLL LZFSE_DLL_EXPORTS)
install(TARGETS lzfse DESTINATION .)
install(FILES "${apple_lzfse_SOURCE_DIR}/LICENSE" DESTINATION . RENAME LZFSE-LICENSE.txt)
